cmake_minimum_required(VERSION 3.16)
project(sovereign_solver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(SOVEREIGN_ENABLE_CUDA "Build optional resident CUDA box-QP backend" OFF)
if(SOVEREIGN_ENABLE_CUDA)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 14)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  target_sources(solver PRIVATE src/cuda_box.cu)
  target_compile_definitions(solver PUBLIC SOVEREIGN_CUDA=1)
  set_target_properties(solver PROPERTIES CUDA_SEPARABLE_COMPILATION ON CUDA_STANDARD 14 CUDA_STANDARD_REQUIRED ON)
endif()
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(solver PUBLIC OpenMP::OpenMP_CXX)
endif()
